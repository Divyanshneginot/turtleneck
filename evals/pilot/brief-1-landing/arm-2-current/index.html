Setup & Integration

Integrate with any AI coding agent in seconds.

Drop Turtleneck into Claude Code, Cursor, Windsurf, Roo, Cline, or direct Agent SDK pipelines. 

# Install Turtleneck design skill globally or per-project
npx turtleneck@latest init --archetype "high-density-starlight"

# Or add directly via your package manager
pnpm add -D @turtleneck/agent-skill
bun add -d @turtleneck/agent-skill

# Run audit against any generated file
npx turtleneck audit ./src/components/dashboard.tsx
// claude_desktop_config.json or any Model Context Protocol host
{
  "mcpServers": {
    "turtleneck-craft": {
      "command": "npx",
      "args": [
        "-y",
        "@turtleneck/mcp-server",
        "--strict-mandate",
        "--archetype=high-density-starlight"
      ]
    }
  }
}
# .cursorrules or .windsurfrules
TURTLENECK_AUTONOMOUS_CRAFT_MANDATE:
  - Archetype: High-Density Starlight (Void #08090e, Surface1 #121520, Surface2 #181c2b, Surface3 #20263a)
  - Timing: 80-120ms cubic-bezier(0.16, 1, 0.3, 1). Disallow transition:all.
  - Interactivity: Enforce 5 states (default, hover, active scale(0.97), focus-visible, disabled).
  - Typography: Strict >= 4:1 tension between display and monospace metadata.
  - Tokens: 8pt grid, WCAG 2.2 AA contrast compliance required on all foregrounds.
import { TurtleneckSkill, ARCHETYPES } from "@turtleneck/agent-skill";
import { AgentPipeline } from "@anthropic-ai/claude-agent-sdk";

const craftSkill = new TurtleneckSkill({
  archetype: ARCHETYPES.HIGH_DENSITY_STARLIGHT,
  enforceTiming: "80ms-120ms",
  activeScale: 0.97,
  contrastTarget: "WCAG_2.2_AA"
});

const agent = new AgentPipeline({
  skills: [craftSkill]
});

// Agent now automatically emits high-density craft compliant UI
await agent.run("Build a deployment manager UI");
